const express = require('express');
const cors = require('cors');
const { GoogleGenAI } = require('@google/genai'); // Gemini SDK-ga rasmiga ah ee 2026

const app = express();
app.use(cors());
app.use(express.json());

// Ku dar API Key-gaaga Gemini halkaan
const ai = new GoogleGenAI({ apiKey: "YOUR_GEMINI_API_KEY" });

app.post('/api/location', async (req, res) => {
    const { userId, latitude, longitude } = req.body;
    console.log(`Xog cusub: User: ${userId} -> Lat: ${latitude}, Long: ${longitude}`);

    try {
        // AI-da ayaa loo dirayaa Isku-duwaha GPS-ka si ay u sharaxdo deegaanka
        const response = await ai.models.generateContent({
            model: 'gemini-2.5-flash',
            contents: `The user is at Latitude: ${latitude}, Longitude: ${longitude} in Mogadishu, Somalia. 
                       Tell them exactly which district, landmark, or zone they are close to based on these coordinates. 
                       Respond in clear, brief Somali language.`
        });

        const aiAnalysis = response.text;

        // Halkaan waxaad ku dhex qori kartaa koodhka ku keydinaya Database-kaaga (sida MongoDB/PostgreSQL)

        res.status(200).json({
            message: "Xogta waa la helay",
            aiAnalysis: aiAnalysis
        });

    } catch (error) {
        console.error("Cillad AI:", error);
        res.status(500).json({ error: "Cillad ayaa ku dhacday nidaamka AI-da." });
    }
});

app.listen(3000, () => console.log('Server-ku wuxuu ka shaqaynayaa Port 3000'));
